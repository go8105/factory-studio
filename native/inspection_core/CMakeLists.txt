cmake_minimum_required(VERSION 3.24)
project(inspection_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(INSPECTION_FETCH_DEPS "Download pinned OpenCV and JSON sources" OFF)
if(INSPECTION_FETCH_DEPS)
  include(FetchContent)
  set(BUILD_LIST core,imgproc CACHE STRING "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  foreach(flag BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_opencv_apps BUILD_JAVA BUILD_opencv_python3 WITH_IPP WITH_ITT WITH_OPENCL WITH_LAPACK WITH_EIGEN OPENCV_ENABLE_NONFREE)
    set(${flag} OFF CACHE BOOL "" FORCE)
  endforeach()
  FetchContent_Declare(opencv URL https://github.com/opencv/opencv/archive/refs/tags/4.13.0.tar.gz
    URL_HASH SHA256=1d40ca017ea51c533cf9fd5cbde5b5fe7ae248291ddf2af99d4c17cf8e13017d DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
    URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(opencv json)
  set(OpenCV_INCLUDE_DIRS "${opencv_SOURCE_DIR}/modules/core/include" "${opencv_SOURCE_DIR}/modules/imgproc/include" "${opencv_BINARY_DIR}" "${CMAKE_BINARY_DIR}")
else()
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
  find_package(nlohmann_json 3.12 REQUIRED)
endif()
add_library(inspection_engine STATIC src/inspection.cpp)
target_include_directories(inspection_engine PUBLIC src)
target_include_directories(inspection_engine SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(inspection_engine PUBLIC nlohmann_json::nlohmann_json PRIVATE opencv_core opencv_imgproc)
if(MSVC)
  target_compile_options(inspection_engine PRIVATE /W4)
else()
  target_compile_options(inspection_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(inspection_core src/main.cpp)
target_link_libraries(inspection_core PRIVATE inspection_engine)
set_target_properties(inspection_core PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_executable(inspection_core_tests tests/core_test.cpp)
target_link_libraries(inspection_core_tests PRIVATE inspection_engine)
add_test(NAME native_core COMMAND inspection_core_tests)
find_program(NODE_EXECUTABLE NAMES node REQUIRED)
add_test(NAME web_native_parity COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/parity.mjs" "$<TARGET_FILE:inspection_core>")
set_tests_properties(web_native_parity PROPERTIES TIMEOUT 120)
